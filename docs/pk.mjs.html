import Sequelize from 'sequelize'
import get from 'lodash-es/get.js'


//主鍵之型別判定、正規化與等值判定之輔助
//
//本模組為[何謂字串型別/有效主鍵/主鍵等值]之唯一擁有者, 其他模組不得另寫同類判定(見CLAUDE_rulebook.md之R-PK)。
//主鍵等值一律以Jet唯一索引之判定為準, 本模組只負責:
//  1. normPk: 依主鍵欄型別正規化主鍵值, 並判定無效、型別不符與不可能存在
//  2. kFast: 快速對應鍵, 只會少判不會多判, 對不上者另以Jet之=查詢判定(insert撞鍵核對與select/delAll之主鍵條件共用)
//  3. castPkFind: 把查詢條件內之主鍵等值類條件改寫為[命中列之序號集合], 序號集合由呼叫端以Jet判定後提供;
//     主鍵以外之條件原樣保留, 交w-orm-reladb之cvFind轉譯(見CLAUDE_rulebook.md之R-FIND)
//
//Jet 4.0之字串等值(已實測): 不分大小寫(含非ASCII)、忽略尾端半形空白、忽略全半形寬度與平片假名、ß視同ss、ς視同σ, 且忽略增補平面字元;
//WHERE不轉型(文字欄以數字比對、數值欄以字串比對皆回報型別不符), INSERT則自動轉型。
//任何以JS仿製之等值鍵皆為近似: 多判使select/delAll命中Jet視為相異之列, 少判則使$ne/$nin把Jet視為同鍵之列當成相異而刪除,
//兩方向於否定條件下皆會誤刪, 故select/delAll之主鍵等值不以近似鍵判定, 而以Jet之=查詢取得命中列(見WOrmMdb.mjs之resolvePkIdx)。


//字串類、整數類、浮點類之model型別(sequelize之type.key), procPk之補值型別檢查與queryMemory之CITEXT改寫共用
//註: sequelize之DataTypes.DOUBLE之key為'DOUBLE PRECISION'(已實查sequelize 6.37.8), 非'DOUBLE'
const TPS_STR = ['STRING', 'TEXT', 'CHAR', 'UUID', 'CITEXT']
const TPS_INT = ['INTEGER', 'BIGINT', 'SMALLINT', 'MEDIUMINT', 'TINYINT']
const TPS_FLT = ['FLOAT', 'DOUBLE PRECISION', 'DOUBLE', 'REAL', 'DECIMAL', 'NUMBER']


//Jet文字欄之長度上限(不計尾端半形空白), 超過者不可能存在於資料表;
//且Jet對超過此長度之字面值於SELECT與DELETE會截斷後比對(UPDATE不截), 送出將命中或刪除以其前255字元為鍵之另一列
const LEN_TEXT_MAX = 255


//記憶體過濾表之原列序號欄名稱前綴, 實際名稱由genInnerKeys附加亂數產生, 以免與使用者之欄位或條件同名
const KEY_WI = '__worm_wi'


//sequelize之運算子Symbol(以Symbol.for註冊, 與w-orm-reladb之cvFind所用者相同)
let Op = Sequelize.Op


//主鍵欄之sequelize原生Symbol運算子所比照之$運算子(同reladb之cvFind於欄位層之處置), 其他原生Symbol原樣留於原欄交cvFind
const MAP_OP_SYM = new Map([
    [Op.eq, '$eq'],
    [Op.ne, '$ne'],
    [Op.gt, '$gt'],
    [Op.gte, '$gte'],
    [Op.lt, '$lt'],
    [Op.lte, '$lte'],
    [Op.in, '$in'],
    [Op.notIn, '$nin'],
    [Op.not, '$not'],
])


//主鍵欄之範圍運算子, 以正規化值留於原欄比對
const OPS_RANGE = ['$gt', '$gte', '$lt', '$lte']


/**
 * 取主鍵欄型別類別
 *
 * @param {Object} attrs 輸入model之rawAttributes
 * @param {String} pk 輸入主鍵欄位名
 * @returns {String} 回傳'str'、'int'、'flt'或'other'
 */
function getPkType(attrs, pk) {
    let k = get(attrs, `${pk}.type.key`, '')
    if (TPS_STR.indexOf(k) >= 0) {
        return 'str'
    }
    if (TPS_INT.indexOf(k) >= 0) {
        return 'int'
    }
    if (TPS_FLT.indexOf(k) >= 0) {
        return 'flt'
    }
    return 'other'
}


/**
 * 拆包裝物件
 *
 * 註: new String(x)與new Number(x)依toString標籤會被判為字串或數字, 但其值不參與===比較(如new String('')!==''),
 * 故先取出原始值, 令其與原始值同處置
 * 註: 以內建之String.prototype.valueOf/Number.prototype.valueOf讀包裝物件之內部值, 不呼叫物件自身之valueOf:
 * 後者可被呼叫端覆寫(每次讀值結果不同), 且僅偽造toStringTag之一般物件並非包裝物件(內建valueOf對其拋錯, 原樣返回而屬無效值)
 *
 * @ignore
 */
function unwrap(v) {
    if (v !== null && typeof v === 'object') {
        try {
            return String.prototype.valueOf.call(v)
        }
        catch (err) {}
        try {
            return Number.prototype.valueOf.call(v)
        }
        catch (err) {}
    }
    return v
}


/**
 * 字串是否無法如實存放於Jet: 含NUL(SQL文字於NUL被截斷)或非良構UTF-16(孤立代理被換為U+FFFD)
 *
 * @ignore
 */
function isUnstorable(s) {
    if (s.indexOf('\u0000') >= 0) {
        return true
    }
    return /[\uD800-\uDBFF](?![\uDC00-\uDFFF])|(?:^|[^\uD800-\uDBFF])[\uDC00-\uDFFF]/.test(s)
}


/**
 * 字串是否超過Jet文字欄上限, 尾端半形空白不計(Jet比對時忽略之, 寫入時亦可容納於上限內)
 *
 * @ignore
 */
function isOverLen(s) {
    return s.replace(/ +$/, '').length > LEN_TEXT_MAX
}


/**
 * 依主鍵欄型別正規化主鍵值
 *
 * 註: kind為'ok'(v為正規化後之值)、'invalid'(未給或非有效值, 依T6處置)、'mismatch'(與主鍵欄型別不符)、'cannot'(不可能存在於資料表)
 * 註: 字串類主鍵欄之有限數字以String(v)處置, 與w-serv-orm一致; 數值類主鍵欄僅轉換標準數字字串(String(Number(s))===s),
 * 其餘字串因Jet之INSERT轉型會改變鍵(如'012'存為12)而視為型別不符; 整數類主鍵欄之非整數因Jet會四捨六入五成雙而視為型別不符
 * 註: ±Infinity、NaN、空字串、布林、物件、陣列、Date、BigInt、null、undefined為無效
 * 註: 超過255字元(不計尾端半形空白)之字串於字串類與其他型別(含model未定義主鍵欄而無從得知型別者)皆為不可能存在
 *
 * @param {*} v 輸入主鍵值
 * @param {String} tp 輸入主鍵欄型別類別, 取自getPkType
 * @returns {Object} 回傳{kind,v}
 */
function normPk(v, tp) {

    //unwrap
    v = unwrap(v)

    //string
    if (typeof v === 'string') {
        if (v === '') {
            return { kind: 'invalid' }
        }
        if (isUnstorable(v)) {
            return { kind: 'cannot' }
        }
        if (tp === 'int' || tp === 'flt') {
            let n = Number(v)
            if (!Number.isFinite(n) || String(n) !== v) {
                return { kind: 'mismatch' }
            }
            if (tp === 'int' && !Number.isInteger(n)) {
                return { kind: 'mismatch' }
            }
            return { kind: 'ok', v: n }
        }
        if (isOverLen(v)) {
            return { kind: 'cannot' }
        }
        return { kind: 'ok', v }
    }

    //number
    if (typeof v === 'number') {
        if (!Number.isFinite(v)) {
            return { kind: 'invalid' }
        }
        if (tp === 'str') {
            return { kind: 'ok', v: String(v) }
        }
        if (tp === 'int' && !Number.isInteger(v)) {
            return { kind: 'mismatch' }
        }
        return { kind: 'ok', v }
    }

    return { kind: 'invalid' }
}


/**
 * 快速對應鍵: 去尾端半形空白並僅將ASCII字母轉小寫
 *
 * 註: 此二者Jet必視為相等(已實測), 故快速鍵相同者必為同一主鍵, 只會少判不會多判;
 * 寬度、假名、非ASCII大小寫等之對應不在此判定, 對不上者另以Jet之=查詢判定
 *
 * @param {String|Number} v 輸入主鍵值
 * @returns {String} 回傳快速鍵字串
 */
function kFast(v) {
    return String(v).replace(/ +$/, '').replace(/[A-Z]/g, (c) => {
        return c.toLowerCase()
    })
}


/**
 * 產生記憶體過濾表之原列序號欄名稱, 附加亂數令其不與使用者之欄位或條件同名
 *
 * @param {Object} attrs 輸入model之rawAttributes
 * @returns {Object} 回傳{wi}
 */
function genInnerKeys(attrs) {
    for (let i = 0; i < 100; i++) {
        let wi = `${KEY_WI}_${Math.random().toString(36).slice(2, 10)}`
        if (get(attrs, wi) === undefined) {
            return { wi }
        }
    }
    throw new Error('can not generate inner keys for memory filter')
}


/**
 * 是否為一般物件: 原型為Object.prototype或null
 *
 * 註: 查詢條件之判定與走訪不得用wsemi之isobj與lodash之each/keys: 具數值length之物件會被當成陣列樣迭代,
 * Symbol鍵(如Op.and)與不可列舉之鍵會被丟棄, 條件因而變成{}而使delAll刪光全表(見CLAUDE_rulebook.md之R-FIND-02)
 *
 * @param {*} v 輸入任意值
 * @returns {Boolean} 回傳是否為一般物件
 */
function isPlainObj(v) {
    if (v === null || typeof v !== 'object') {
        return false
    }
    let p = Object.getPrototypeOf(v)
    return p === Object.prototype || p === null
}


/**
 * 設定自有屬性, 以defineProperty建構, 令鍵為__proto__時亦為自有鍵而不改原型(交cvFind拒收)
 *
 * @ignore
 */
function putProp(o, k, v) {
    Object.defineProperty(o, k, { value: v, enumerable: true, writable: true, configurable: true })
}


/**
 * 物件是否為運算子物件(一般物件且含任一$開頭之鍵或Symbol鍵); 否則視為一般值(如{}), 屬無效主鍵值
 *
 * @ignore
 */
function isOpObj(c) {
    if (!isPlainObj(c)) {
        return false
    }
    return Reflect.ownKeys(c).some((k) => {
        return typeof k === 'symbol' || k[0] === '$'
    })
}


/**
 * 改寫查詢條件內之主鍵條件
 *
 * 註: 主鍵等值類條件(隱含等值、陣列值、$eq、$ne、$in、$nin, 及其下之$not)改寫為原列序號欄wi之$in/$nin,
 * 序號集合以佔位陣列置於條件內並列於回傳之slots(各含正規化後之主鍵值vals與佔位陣列is), 呼叫端以Jet判定命中列後以fillPkSlots就地填入,
 * 故命中判定與selectByPk、del等一致, 不經任何近似鍵
 * 註: 單趟走訪, 每個條件值只讀一次: 原先之兩趟呼叫(先收集主鍵值、解析後再改寫)於兩趟讀值不一時(覆寫valueOf之包裝物件、陷阱會變之Proxy),
 * 第二趟之值未經收集而查無序號, $ne/$nin之條件因此消失而使delAll刪光全表(run02外審opus、sol)
 * 註: 主鍵欄只認鍵與pk完全相符者(與reladb之cvFind之嚴格鍵一致), 大小寫或尾端空白不同之近似鍵原樣交cvFind拒收
 * 註: 範圍運算子($gt、$gte、$lt、$lte)以正規化值留於原欄比對(記憶體sqlite之語意, 為近似), $regex等其餘運算子原樣留於原欄;
 * sequelize原生Symbol運算子之Op.eq/ne/gt/gte/lt/lte/in/notIn/not比照對應之$運算子, 其他原生Symbol原樣留於原欄
 * 註: 肯定語境之無效、型別不符、不可能存在之值不命中(範圍運算子亦同, 不留null給cvFind);
 * 否定語境($ne、$nin、$not之內, 以及文件層$not、Op.not與$nor之內)之無效或型別不符之值一律拋出:
 * 依[無效值不命中]推得之否定為[全部命中], 用於delAll即清空全表, 而此類值多為呼叫端之錯誤(如變數為undefined);
 * 修正前sequelize對undefined亦為reject, 故拋出不使既有呼叫改為刪光全表; 不可能存在之值於否定語境依其語意為必不相等
 * 註: 主鍵欄之$not(含Op.not)整體取反NOT(A∧B), 內層以否定語境遞迴改寫, 各$not各自成一子式而不互相覆寫; 值為純量或陣列者即$nin
 * 註: 主鍵欄之RegExp值視為$regex(主控裁示Q4); 非運算子物件(如{})視為主鍵值而非條件, 屬無效值, 不得因其無運算子而使主鍵條件消失(修正初版即因此使delAll({id:{}})刪光全表)
 * 註: 主鍵所衍生之條件以{[Op.and]:[其餘, ...主鍵條件]}包覆(其餘為空時省略), 不併入呼叫端之$and(呼叫端之空$and仍由cvFind拒收)
 * 註: 走訪以Reflect.ownKeys(含Symbol鍵與不可列舉之鍵)並以defineProperty建構, 不用lodash之each: 具數值length之物件會被當成陣列樣迭代而使條件消失;
 * 非一般物件原樣返回(交cvFind拒收); 主鍵以外之欄位、運算子與值原樣保留交cvFind檢核與轉譯
 * 註: 文件層之$not(頂層與$and/$or/$nor/Op.not之元素內, 值為一般物件)轉為Op.not子式(主控裁示Q2: 本輪保留1.0.35之支援; reladb之cvFind對其為reject)
 * 註: 序號條件不產生sequelize之literal(空集合以不存在之序號-1表示), 令cvFind所用之sequelize與記憶體sqlite者不同份時, 主鍵條件不因literal無法辨識而拒收
 *
 * @param {Object} find 輸入mongo風格之查詢條件, 一般物件(呼叫端已處理非物件與陣列)
 * @param {String} pk 輸入主鍵欄位名, 給null則不辨識主鍵條件(model無此屬性時, 令主鍵名之鍵交cvFind以嚴格鍵拒收)
 * @param {String} tp 輸入主鍵欄型別類別
 * @param {String} wi 輸入原列序號欄名稱, 取自genInnerKeys
 * @returns {Object} 回傳{find:改寫後之查詢條件, slots:[{vals:正規化後之主鍵值陣列, is:序號佔位陣列}]}
 */
function castPkFind(find, pk, tp, wi) {

    //slots, 序號集合之佔位
    let slots = []

    //err, 否定語境之無效值
    function err(v) {
        return new Error(`invalid ${pk} in negative condition: ${String(v)}`)
    }

    //normVals, 多值之正規化(純量視為單元素陣列); 肯定語境之無效、型別不符與不可能存在者略過(不命中); 否定語境之無效或型別不符拋出
    function normVals(vs, neg) {
        if (!Array.isArray(vs)) {
            vs = [vs]
        }
        let r = []
        for (let k = 0; k < vs.length; k++) { //逐索引, 陣列之洞即undefined(無效值)
            let n = normPk(vs[k], tp)
            if (n.kind === 'ok') {
                r.push(n.v)
            }
            else if (neg && (n.kind === 'invalid' || n.kind === 'mismatch')) {
                throw err(vs[k])
            }
        }
        return r
    }

    //slot, 序號佔位陣列, 由fillPkSlots就地填入
    function slot(vals) {
        let s = { vals, is: [] }
        slots.push(s)
        return s.is
    }

    //cIn, cNin, 序號條件; 無有效值之$in以不存在之序號表示必不命中, 無有效值之$nin為恆成立而不加條件
    function cIn(vals) {
        return { [wi]: { $in: vals.length > 0 ? slot(vals) : [-1] } }
    }
    function cNin(vals) {
        if (vals.length === 0) {
            return null
        }
        return { [wi]: { $nin: slot(vals) } }
    }

    //rangeVal, 範圍運算子之值: ok回{v}; 否定語境之無效或型別不符拋出; 其餘(肯定語境之無效、型別不符, 與不可能存在)回{no:true}表該子式不命中
    function rangeVal(v, neg) {
        let r = normPk(v, tp)
        if (r.kind === 'ok') {
            return { v: r.v }
        }
        if (neg && (r.kind === 'invalid' || r.kind === 'mismatch')) {
            throw err(v)
        }
        return { no: true }
    }

    //push
    function push(conds, c) {
        if (c) {
            conds.push(c)
        }
    }

    //nameOf, 運算子名稱: Symbol鍵取其比照之$運算子(非比照者為null), 字串鍵原樣
    function nameOf(k) {
        if (typeof k === 'symbol') {
            return MAP_OP_SYM.get(k) || null
        }
        return k
    }

    //putPk, 以主鍵欄為鍵之條件物件
    function putPk(v) {
        let o = {}
        putProp(o, pk, v)
        return o
    }

    //castNot, 主鍵欄之$not(含Op.not): 運算子物件整體取反NOT(A∧B), 內層以否定語境改寫(內層之$not遞迴), 每個$not各自成一子式
    function castNot(v, conds) {
        if (v instanceof RegExp) {
            conds.push(putPk({ $not: v }))
            return
        }
        if (!isOpObj(v)) {
            push(conds, cNin(normVals(v, true)))
            return
        }
        let r = castCond(v, true)
        let subs = [...r.conds]
        if (Reflect.ownKeys(r.org).length > 0) {
            subs.push(putPk(r.org))
        }
        if (subs.length === 0) {
            subs.push({ [wi]: { $nin: [-1] } }) //內層無子式即恆真(如$nin:[]), 以恆真之序號條件代替{}, 令邏輯陣列非空且不產生literal
        }
        conds.push({ [Op.not]: { [Op.and]: subs } })
    }

    //castCond, 主鍵欄之條件, 回傳{conds:[序號條件或取反子式], org:{原欄條件}}
    function castCond(c, neg) {
        let conds = []
        let org = {}
        if (c instanceof RegExp) {
            putProp(org, '$regex', c)
            return { conds, org }
        }
        if (!isOpObj(c)) {
            push(conds, cIn(normVals(c, neg)))
            return { conds, org }
        }
        for (let k of Reflect.ownKeys(c)) {
            let v = c[k]
            let nm = nameOf(k)
            if (nm === '$eq') {
                push(conds, cIn(normVals([v], neg))) //單值: 陣列值屬無效值
            }
            else if (nm === '$in') {
                push(conds, cIn(normVals(v, neg)))
            }
            else if (nm === '$ne') {
                push(conds, cNin(normVals([v], true)))
            }
            else if (nm === '$nin') {
                push(conds, cNin(normVals(v, true)))
            }
            else if (nm === '$not') {
                castNot(v, conds)
            }
            else if (OPS_RANGE.indexOf(nm) >= 0) {
                let r = rangeVal(v, neg)
                if (r.no) {
                    push(conds, cIn([]))
                }
                else {
                    putProp(org, k, r.v)
                }
            }
            else {
                putProp(org, k, v)
            }
        }
        //org只剩視同未給之$options(undefined、null、'', 同reladb之R6)而已有主鍵子式者, 移除之:
        //非主鍵之{f:{$eq:x,$options:undefined}}為合法, 主鍵之等值被抽為序號條件後若留下{$options:undefined}, cvFind會以無運算子拒收(run02外審sol)
        let ks = Reflect.ownKeys(org)
        if (conds.length > 0 && ks.length === 1 && ks[0] === '$options' && (org.$options === undefined || org.$options === null || org.$options === '')) {
            org = {}
        }
        return { conds, org }
    }

    //castObj, neg為是否位於否定語境(文件層$not、Op.not與$nor之內)
    function castObj(o, neg) {
        if (!isPlainObj(o)) {
            return o
        }
        let rest = {}
        let ands = []
        for (let k of Reflect.ownKeys(o)) {
            let v = o[k]
            if (k === pk) {
                let { conds, org } = castCond(v, neg)
                ands.push(...conds)
                if (Reflect.ownKeys(org).length > 0) {
                    putProp(rest, k, org)
                }
            }
            else if ((k === '$and' || k === '$or' || k === Op.and || k === Op.or) && Array.isArray(v)) {
                putProp(rest, k, v.map((x) => { //map保留陣列之洞, 交cvFind拒收
                    return castObj(x, neg)
                }))
            }
            else if (k === '$nor' && Array.isArray(v)) {
                putProp(rest, k, v.map((x) => {
                    return castObj(x, true)
                }))
            }
            else if (k === Op.not && isPlainObj(v)) {
                putProp(rest, k, castObj(v, true))
            }
            else if (k === '$not' && isPlainObj(v)) {
                ands.push({ [Op.not]: castObj(v, true) })
            }
            else {
                putProp(rest, k, v)
            }
        }
        if (ands.length === 0) {
            return rest
        }
        if (Reflect.ownKeys(rest).length === 0) {
            return { [Op.and]: ands } //其餘為空時不放{}(cvFind轉為literal之恆真)
        }
        return { [Op.and]: [rest, ...ands] }
    }

    //find, 非物件與陣列視為未給(系列現況保留; 呼叫端findAll已先處理), 非一般物件原樣返回交cvFind拒收
    if (find === null || typeof find !== 'object' || Array.isArray(find)) {
        return { find: {}, slots }
    }
    return { find: castObj(find, false), slots }
}


/**
 * 依各主鍵值命中之序號就地填入castPkFind之序號佔位陣列
 *
 * 註: 各佔位為其主鍵值命中序號之聯集(遞增); 聯集為空者填入不存在之序號-1($in即不命中、$nin即恆成立), 不留空陣列(cvFind對空集合產生literal)
 *
 * @param {Array} slots 輸入castPkFind回傳之slots
 * @param {Function} idxOf 輸入函數, 給正規化後之主鍵值回傳命中列之序號陣列
 */
function fillPkSlots(slots, idxOf) {
    for (let s of slots) {
        let set = new Set()
        for (let v of s.vals) {
            for (let i of idxOf(v)) {
                set.add(i)
            }
        }
        let is = [...set].sort((a, b) => a - b)
        s.is.push(...(is.length > 0 ? is : [-1]))
    }
}
